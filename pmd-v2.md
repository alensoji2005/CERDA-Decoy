```python
# src/decoy/perturbation.py

from enum import Enum
import random
import time

class Regime(Enum):
    R0_BASELINE = "baseline"        # No artificial delay
    R1_SLOWDOWN = "slowdown"        # +100ms fixed delay before response
    R2_JITTER = "jitter"            # Random delay 50-200ms
    R3_FS_LATENCY = "fs_latency"    # +200ms for filesystem operations only

def apply_perturbation(regime: Regime, operation_type: str) -> None:
    """
    Apply delay based on regime before responding to command.
    
    Args:
        regime: Current perturbation regime
        operation_type: "filesystem" or "network" or "command"
    """
    delay_ms = 0
    if regime == Regime.R1_SLOWDOWN:
        delay_ms = 100
    elif regime == Regime.R2_JITTER:
        delay_ms = random.randint(50, 200)
    elif regime == Regime.R3_FS_LATENCY and operation_type == "filesystem":
        delay_ms = 200
    
    if delay_ms > 0:
        time.sleep(delay_ms / 1000.0)
```

- Each session gets a **deterministic schedule** based on session seed
- Schedule format: list of `(command_index_start, regime)` tuples
- Default schedule (configurable):
    - Commands 0-4: R0
    - Commands 5-9: R1
    - Commands 10-14: R2
    - Commands 15+: R3

**Critical:** The schedule must be logged to `regime_schedule.csv` for each session.

```python
import asyncio
from typing import Optional
from .pseudo_fs import PseudoFilesystem
from .perturbation import Regime
from .telemetry import TelemetryLogger

class DecoyServer:
    def __init__(self, config: dict, telemetry_logger: TelemetryLogger):
        """
        Initialize listeners, pseudo-filesystem, and perturbation engine.
        
        Args:
            config: Configuration dict with ssh_port, http_port, default_schedule
            telemetry_logger: TelemetryLogger instance for writing CSV
        """
        self.config = config
        self.telemetry_logger = telemetry_logger
        self.sessions: dict = {}  # session_id -> PseudoFilesystem
    
    async def start(self) -> None:
        """Start TCP listeners on configured ports."""
        ssh_server = await asyncio.start_server(
            self._handle_connection,
            self.config['ssh_host'],
            self.config['ssh_port']
        )
        http_server = await asyncio.start_server(
            self._handle_connection,
            self.config['http_host'],
            self.config['http_port']
        )
        await asyncio.gather(ssh_server.serve_forever(), http_server.serve_forever())
    
    async def _handle_connection(
        self,
        reader: asyncio.StreamReader,
        writer: asyncio.StreamWriter
    ) -> None:
        """Handle a single attacker session."""
        session_id = str(uuid.uuid4())
        pseudo_fs = PseudoFilesystem(session_id)
        self.sessions[session_id] = pseudo_fs
        
        # Send welcome message
        welcome = f"Welcome to {pseudo_fs.current_path}#\n"
        writer.write(welcome.encode())
        await writer.drain()
        
        command_index = 0
        last_timestamp_ns = time.monotonic_ns()
        
        while True:
            try:
                data = await asyncio.wait_for(reader.read(1024), timeout=30.0)
                if not data:
                    break
                
                command = data.decode('utf-8', errors='ignore').strip()
                current_timestamp_ns = time.monotonic_ns()
                delta_t_ns = current_timestamp_ns - last_timestamp_ns
                last_timestamp_ns = current_timestamp_ns
                
                # Determine current regime based on schedule
                regime = self._get_regime_for_command(command_index)
                
                # Log telemetry BEFORE responding
                self.telemetry_logger.log(
                    session_id=session_id,
                    timestamp_ns=current_timestamp_ns,
                    source_ip=writer.get_extra_info('peername')[^0],
                    protocol="SSH" if writer.get_extra_info('sockname')[^1] == self.config['ssh_port'] else "HTTP",
                    command_index=command_index,
                    command_raw=command,
                    command_root=command.split()[^0] if command else "",
                    current_path=pseudo_fs.current_path,
                    regime=regime.value,
                    delta_t_ns=delta_t_ns,
                    response_sent=True
                )
                
                # Apply perturbation delay
                apply_perturbation(regime, "command")
                
                # Execute command in pseudo-filesystem
                response = self._execute_command(pseudo_fs, command)
                
                # Send response
                writer.write(response.encode())
                await writer.drain()
                
                command_index += 1
                
            except asyncio.TimeoutError:
                break
            except Exception as e:
                # Log error but continue session
                break
        
        writer.close()
        await writer.wait_closed()
    
    def _get_regime_for_command(self, command_index: int) -> Regime:
        """Get regime for given command index based on deterministic schedule."""
        for start_idx, regime in reversed(self.config['default_schedule']):
            if command_index >= start_idx:
                return regime
        return Regime.R0_BASELINE
    
    def _execute_command(self, pseudo_fs: PseudoFilesystem, command: str) -> str:
        """Execute command in pseudo-filesystem (no actual OS calls)."""
```

#### 5.1.4 CERF Generation (Patent Core)

**Locality-Sensitive Hashing (LSH):**

```python
def compute_cerf(embedding: np.ndarray, num_hashes: int = 64) -> str:
    """
    Compute Counterfactual Execution-Rhythm Fingerprint.
    
    Args:
        embedding: 32-dim numpy array
        num_hashes: Number of hash bits (default 64)
    
    Returns:
        Hex string of CERF (e.g., "a3f5c9d2...")
    """
    # Generate random projection matrix (fixed seed for reproducibility)
    projection_matrix = np.random.RandomState(seed=42).randn(32, num_hashes)
    
    # Project embedding to hash space
    projections = np.dot(embedding, projection_matrix)
    
    # Quantize to binary (1 if > 0, else 0)
    binary_hash = (projections > 0).astype(int)
    
    # Convert to hex string
    cerf = "".join([format(int("".join(map(str, binary_hash[i:i+4])), 2), 'x')
                    for i in range(0, num_hashes, 4)])
    return cerf
```

**Critical:** The CERF must be deterministic (same embedding → same hash) and reproducible across runs.

#### 5.2.1 CERF Engine Class

```python
# src/analytics/cerf_engine.py

import pandas as pd
import numpy as np
from typing import Dict, List, Tuple
from .sanitization import sanitize_commands
from .features import extract_response_features
from .embedding import EmbeddingModel
from .clustering import cluster_by_cerf

class CERFEngine:
    def __init__(self, telemetry_path: str, model_dir: str = "models/"):
        """
        Initialize with paths to telemetry and model storage.
        
        Args:
            telemetry_path: Path to telemetry.csv
            model_dir: Directory to save/load embedding models
        """
        self.telemetry_path = telemetry_path
        self.model_dir = model_dir
        self.embedding_model = EmbeddingModel(model_dir)
    
    def load_telemetry(self) -> pd.DataFrame:
        """Load and sanitize telemetry data."""
        df = pd.read_csv(self.telemetry_path)
        df = sanitize_commands(df)
        return df
    
    def extract_features(self, telemetry_df: pd.DataFrame) -> Dict[str, np.ndarray]:
        """Compute feature vectors for all sessions."""
        return extract_response_features(telemetry_df)
    
    def train_embedding_model(self, feature_vectors: Dict[str, np.ndarray]) -> None:
        """Fit scaler and PCA, save to disk."""
        self.embedding_model.train(feature_vectors)
    
    def compute_cerf_for_session(self, session_id: str, feature_vector: np.ndarray) -> str:
        """Compute CERF for a single session."""
        embedding = self.embedding_model.transform(feature_vector)
        cerf = compute_cerf(embedding)
        return cerf
    
    def cluster_sessions(self, cerfs: Dict[str, str]) -> Dict[str, str]:
        """Group sessions by CERF, assign threat_actor_id."""
        return cluster_by_cerf(cerfs)
    
    def run_pipeline(self) -> pd.DataFrame:
        """Execute full pipeline: load → features → CERF → cluster → output."""
        telemetry_df = self.load_telemetry()
        feature_vectors = self.extract_features(telemetry_df)
        
        # Train embedding model if not already trained
        if not self.embedding_model.is_trained():
            self.train_embedding_model(feature_vectors)
        
        # Compute CERFs
        cerfs = {}
        for session_id, fv in feature_vectors.items():
            cerf = self.compute_cerf_for_session(session_id, fv)
            cerfs[session_id] = cerf
        
        # Cluster sessions
        threat_actor_ids = self.cluster_sessions(cerfs)
        
        # Build output DataFrame
        output_df = pd.DataFrame([
            {
                "session_id": session_id,
                "source_ip": telemetry_df[telemetry_df['session_id'] == session_id]['source_ip'].iloc[^0],
                "cerf": cerf,
                "threat_actor_id": threat_actor_ids[session_id],
                "embedding_norm": np.linalg.norm(self.embedding_model.transform(fv)),
                "num_commands": len(telemetry_df[telemetry_df['session_id'] == session_id]),
                "dominant_regime": telemetry_df[telemetry_df['session_id'] == session_id]['regime'].mode().iloc[^0]
            }
            for session_id, (cerf, fv) in zip(cerfs.keys(), feature_vectors.values())
        ])
        
        return output_df
```

## 9. Risk Mitigation

### Risk 1: Perturbation Schedule Too Aggressive

**Symptom:** Attackers detect honeypot and disconnect early
**Mitigation:**

- Start with conservative delays (R1: 50ms, R2: 25-100ms)
- Add "stealth mode" config option to disable R2/R3


### Risk 2: CERF Overfitting to Synthetic Data

**Symptom:** Poor attribution on real-world traffic
**Mitigation:**

- Test with public honeypot datasets (e.g., Cowrie logs)
- Use cross-validation on embedding model


### Risk 3: Resource Exhaustion on Edge Devices

**Symptom:** RAM > 500 MB, slow dashboard
**Mitigation:**

- Batch CERF computation (every 60 seconds)
- Downsample telemetry for visualization

## 10. Patent Claim Draft (For Attorney Review)

**Independent Claim (Conceptual):**

> A computer-implemented method for attributing distributed network intrusions to a singular automated threat actor across discrete network addresses, comprising:
>
> (a) operating a decoy system that interacts with attackers via emulated protocols;
>
> (b) applying a sequence of controlled timing and state perturbations to the decoy interaction according to a deterministic schedule;
>
> (c) recording attacker commands and inter-command delays under each perturbation regime;
>
> (d) constructing a response model that characterizes how the attacker's command timing varies as a function of the applied perturbations;
>
> (e) generating a counterfactual execution-rhythm fingerprint by applying a locality-sensitive hashing function to an embedding derived from the response model and command structure;
>
> (f) attributing sessions from different network addresses to the same automated threat actor when their fingerprints satisfy a similarity criterion.

**Dependent Claims (Examples):**

1. The method of claim 1, wherein the controlled timing perturbations comprise fixed delays, randomized jitter, and filesystem operation latencies.
2. The method of claim 1, wherein the response model includes median inter-command delays per regime, ratios of delays between regimes, and adaptation speed metrics.
3. The method of claim 1, wherein the embedding is derived from principal component analysis (PCA) of a feature vector combining timing features and command AST features.
4. The method of claim 1, wherein the similarity criterion is exact hash match or Hamming distance ≤ 4 bits.
5. The method of claim 1, wherein applying perturbations comprises a reinforcement learning agent dynamically selecting a next perturbation regime to maximize information gain regarding threat actor response characteristics.
6. The method of claim 1, wherein controlled timing perturbations comprise manipulating network socket-level TCP acknowledgments and window sizes to extract operating system network-stack signatures.
7. The method of claim 1, wherein the decoy system generates synthetic, context-aware filesystem responses dynamically utilizing a local language model.
8. The method of claim 1, wherein the response model embedding is trained using a sequence-to-vector recurrent neural network or Transformer model over inter-command delay sequences.
